# Box3D physics PoC scene.
# The authored scenegraph is authoritative; glTF files are pure geometry sources.
# Transforms accumulate down the hierarchy (parent world * local).
#
# Grammar:
#   gltf <name> <path>                                 register a geometry source
#   group <var> | transform <var>                      group / transform node
#   translate/rotate/scale <var> ...                   compose onto a transform node
#   mesh <var> <gltfName>                              leaf referencing geometry
#   add-child <child> <parent>                         wire hierarchy
#   root <var>                                         set the graph root
#   rigidbody <var> <static|dynamic|kinematic> <box|sphere|capsule|hull> [density]

# --- geometry sources ---
gltf g_ground  ../physics_models/box3d_ground.gltf
gltf g_cube    ../physics_models/box3d_cube.gltf
gltf g_sphere  ../physics_models/box3d_sphere.gltf
gltf g_capsule ../physics_models/box3d_capsule.gltf
gltf g_hull    ../physics_models/box3d_hull.gltf
gltf g_body    ../physics_models/wip_robot_without_negate.glb
gltf g_head    ../physics_models/wip_robot_head_without_negate.glb

# --- scene root ---
group root

# --- static ground: unit cube scaled to a wide flat slab, top surface at y = 0 ---
transform t_ground
translate t_ground 0 -1 0
scale t_ground 20 1 20
mesh m_ground g_ground
add-child m_ground t_ground
add-child t_ground root
rigidbody m_ground static box

# --- dynamic box, dropped and slightly tilted so it tumbles ---
transform t_box
translate t_box -6 9 0
rotate t_box 25 1 0 1
mesh m_box g_cube
add-child m_box t_box
add-child t_box root
rigidbody m_box dynamic box 1.0

# --- dynamic sphere ---
transform t_sphere
translate t_sphere -2 11 0
mesh m_sphere g_sphere
add-child m_sphere t_sphere
add-child t_sphere root
rigidbody m_sphere dynamic sphere 1.0

# --- dynamic capsule, tilted so it topples ---
transform t_capsule
translate t_capsule 2 10 0
rotate t_capsule 30 0 0 1
mesh m_capsule g_capsule
add-child m_capsule t_capsule
add-child t_capsule root
rigidbody m_capsule dynamic capsule 1.0

# --- dynamic convex hull (icosahedron 'rock') ---
transform t_hull
translate t_hull 6 12 0
rotate t_hull 20 1 1 0
mesh m_hull g_hull
add-child m_hull t_hull
add-child t_hull root
rigidbody m_hull dynamic hull 1.0

# --- self-assembly pair --- modelled in mm, hence the 0.1
transform t_body
translate t_body 12 3 0
scale t_body 0.1 0.1 0.1
mesh m_body g_body
add-child m_body t_body
add-child t_body root
rigidbody m_body dynamic hull 1.0
# north base on the flat collar top (y=25.81), 4 magnets on a 16mm square. Odd polarity in one
# corner is what makes a mismatched orientation attract on only some magnets.
magnet m_body -4.25 25.81  4.25 0 1 0 s
magnet m_body  4.25 25.81  4.25 0 1 0 n
magnet m_body -4.25 25.81 -4.25 0 1 0 n
magnet m_body  4.25 25.81 -4.25 0 1 0 n

# beside the body, not above it, so a horizontal force isn't fighting gravity
transform t_head
translate t_head 16 2 0
scale t_head 0.1 0.1 0.1
mesh m_head g_head
add-child m_head t_head
add-child t_head root
rigidbody m_head dynamic hull 1.0
# provisional: this head has no flat base modelled yet, so these sit on the sphere's underside.
# south base, so the odd magnet mirrors to the opposite corner.
magnet m_head  2.91 1.50  5.56 0 -1 0 n
magnet m_head -5.59 1.50  5.56 0 -1 0 s
magnet m_head  2.91 1.50 -2.94 0 -1 0 s
magnet m_head -5.59 1.50 -2.94 0 -1 0 s

root root
